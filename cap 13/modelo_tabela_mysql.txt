USE cl203420;

CREATE TABLE exemplo (
    id            INT AUTO_INCREMENT PRIMARY KEY,  -- número inteiro que cresce sozinho (1, 2, 3...)
    nome          VARCHAR(100) NOT NULL,           -- texto curto, até 100 letras, obrigatório
    email         VARCHAR(150) UNIQUE,             -- texto, não pode repetir
    senha         VARCHAR(255),                    -- 255 para caber o password_hash()
    descricao     TEXT,                            -- texto longo (sem limite prático)
    sigla         CHAR(2),                         -- texto de tamanho fixo (ex.: SP, RJ)
    quantidade    INT DEFAULT 0,                   -- número inteiro, começa com 0
    preco         DECIMAL(10,2),                   -- dinheiro: 10 dígitos, 2 depois da vírgula
    peso          FLOAT,                           -- número com vírgula (aproximado)
    ativo         BOOLEAN DEFAULT TRUE,            -- verdadeiro/falso (guarda 1 ou 0)
    nascimento    DATE,                            -- data: '2026-10-19'
    horario       TIME,                            -- hora: '14:30:00'
    criado_em     DATETIME DEFAULT CURRENT_TIMESTAMP, -- data e hora, preenchida sozinha
    perfil        ENUM('ADM', 'FUNC')              -- só aceita os valores da lista
);

-- =====================================================
-- RESUMO: QUAL TIPO USAR?
-- =====================================================
-- Inteiro (idade, quantidade, id) ........ INT
-- Dinheiro (preço, salário) ............. DECIMAL(10,2)
-- Número com vírgula (peso, nota) ....... FLOAT
-- Texto curto (nome, email) ............. VARCHAR(tamanho)
-- Texto longo (descrição, comentário) ... TEXT
-- Sim/Não ............................... BOOLEAN
-- Data / Hora / Data e hora ............. DATE / TIME / DATETIME
-- Lista fixa de opções .................. ENUM('A', 'B')
